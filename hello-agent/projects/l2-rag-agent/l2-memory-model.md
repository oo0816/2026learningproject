│  │  Layer 1    │    │  Layer 2     │    │  Layer 3     │   │

│  │   messages  │    │  key-value   │    │  list[dict]  │   │
│  ├─────────────┤    ├──────────────┤    ├──────────────┤   │
│  │ 生命周期:    │    │ 生命周期:     │    │ 生命周期:     │   │
│  │  单次会话    │    │  单次会话      │    │  跨会话持久    │   │
│  ├─────────────┤    ├──────────────┤    ├──────────────┤   │

│  │  限制决定    │    │  key-value    │    │  可无限增长   │   │

│  │ 写入方式:    │    │ 写入方式:      │    │ 写入方式:     │   │
│  │  append到   │    │  REMEMBER   │    │  显式存储      │   │
│  │  messages   │    │  指令提取     │    │  或自动摘要    │   │
│  ├─────────────┤    ├──────────────┤    ├──────────────┤   │
│  │ 检索方式:    │    │ 检索方式:      │    │ 检索方式:     │   │

│  └─────────────┘    └──────────────┘    └──────────────┘   │
│                                                             │
│  ┌─────────────────────────────────────────────────────┐   │
│  │              数据流方向                               │   │
│  │                                                     │   │
│  │  messages ──(提取关键信息)──→ session_store           │   │
│  │  session_store ──(会话结束, 重要信息)──→ long_term    │   │
│  │  long_term ──(检索后注入)──→ system prompt            │   │
│  └─────────────────────────────────────────────────────┘   │
